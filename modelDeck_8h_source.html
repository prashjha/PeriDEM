191 void readFromJson(const json &j) {
192 if (j.empty())
193 return;
194
195 d_dim = j.at("Dimension");
196 d_tFinal = j.at("Final_Time");
197 d_Nt = j.at("Time_Steps");
198 d_spatialDiscretization = j.at("Discretization_Type").at("Spatial");
199 d_timeDiscretization = j.at("Discretization_Type").at("Time");
200 if (d_timeDiscretization == "central_difference" or d_timeDiscretization == "velocity_verlet")
201 d_simType = "explicit";
202 d_populateElementNodeConnectivity = j.value("Populate_ElementNodeConnectivity", true);
203 d_quadOrder = j.value("Quad_Approximation_Order", size_t(2));
204 d_particleSimType = j.value("Particle_Sim_Type", "Multi_Particle");
205 d_mpiStrategy = j.value("MPI_Strategy", "auto");
206 d_bondBreak = j.value("Bond_Break", "tension");
207 d_selfContact = j.value("Self_Contact", "broken_bond_kn");
208 d_wallContact = j.value("Wall_Contact", "meshed");
209 d_seed = j.value("Seed", 0);
210
211 d_rigidParticles.clear();
212 if (j.find("Rigid_Particles") != j.end()) {
213 for (const auto &r : j.at("Rigid_Particles")) {
214 const double mass = r.value("Mass", -1.);
215 if (!(mass > 0.)) {
216 std::cerr << "Error: Model.Rigid_Particles entries need Mass > 0.\n";
217 exit(1);
218 }
219 d_rigidParticles.emplace_back(r.at("Id").get<size_t>(), mass);
220 }
221 }
222
223 if (d_mpiStrategy != "auto" && d_mpiStrategy != "none" &&
224 d_mpiStrategy != "particle" && d_mpiStrategy != "dof") {
225 std::cerr << "Error: Model.MPI_Strategy must be auto|none|particle|dof.\n";
226 exit(1);
227 }
228 if (d_bondBreak != "tension" && d_bondBreak != "absolute_stretch") {
229 std::cerr << "Error: Model.Bond_Break must be tension|absolute_stretch.\n";
230 exit(1);
231 }
232 if (d_selfContact != "broken_bond_kn" &&
233 d_selfContact != "reference_gap" && d_selfContact != "none") {
234 std::cerr << "Error: Model.Self_Contact must be "
235 "broken_bond_kn|reference_gap|none.\n";
236 exit(1);
237 }
238 if (d_wallContact != "meshed" && d_wallContact != "analytical_plane") {
239 std::cerr << "Error: Model.Wall_Contact must be "
240 "meshed|analytical_plane.\n";
241 exit(1);
242 }
243
244 if (std::abs(d_tFinal) < 1.0E-10 or d_Nt <= 0) {
245 std::cerr << "Error: Check Final_Time and Time_Steps data.\n";
246 exit(1);
247 }
248
249 d_dt = d_tFinal / d_Nt;
250 }
