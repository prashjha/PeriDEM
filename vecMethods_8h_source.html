40inline size_t maxIndex(const std::vector<T> &data) {
41
42 // initialize original index locations
43 std::vector<size_t> idx(data.size());
44 std::iota(idx.begin(), idx.end(), 0);
45
46 std::stable_sort(idx.begin(), idx.end(),
47 [&data](size_t i1, size_t i2) {return data[i1] > data[i2];});
48
49 return idx[0];
50};

118template <typename T>
119inline size_t maxIndex(const std::vector<T> &data,
120 size_t data_start, size_t data_end) {
121
122 if (data.size() == 0) {
123 throw std::runtime_error(
124 util::io::Msg()
125 << "Error: maxIndex() is called with data of size " << data.size()
126 << ".\n");
127 }
128
129 if (data_end == 0 or data_end > data.size()) {
130 throw std::runtime_error(
131 util::io::Msg()
132 << "Error: maxIndex() data_end = " << data_end
133 << " is not valid for the data of size " << data.size()
134 << ".\n");
135 }
136
137 if (data_start > data.size() - 1) {
138 throw std::runtime_error(
139 util::io::Msg()
140 << "Error: maxIndex() data_start = " << data_start
141 << " is not valid for the data of size " << data.size()
142 << ".\n");
143 }
144
145 // initialize original index locations
146 std::vector<size_t> idx(data_end - data_start);
147 std::iota(idx.begin(), idx.end() , 0);
148
149 std::stable_sort(idx.begin(), idx.end(),
150 [&data, &data_start](size_t i1, size_t i2)
151 {return data[i1 + data_start] > data[i2 + data_start];});
152
153 return idx[0] + data_start;
154};

311 if (unit == "microseconds")

319 util::io::Msg()

394template <class T>
395inline T l2Dist(const std::vector<T> &x1, const std::vector<T> &x2) {
396
397 if (x1.size() != x2.size())
398 throw std::invalid_argument("size of two vectors not matching");
399
400 T sum = 0.;
401 for (size_t i=0; i<x1.size(); i++)
402 sum += (x1[i] - x2[i]) * (x1[i] - x2[i]);
403
404 return std::sqrt(sum);
405}
