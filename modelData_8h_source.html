515 void setFix(size_t i, const unsigned int &dof,
516 const bool &flag) {
517 // to set i^th bit as true of integer a,
518 // a |= 1UL << (i % 8)
519
520 // to set i^th bit as false of integer a,
521 // a &= ~(1UL << (i % 8))
522
523 flag ? (d_fix[i] |= 1UL << dof) : (d_fix[i] &= ~(1UL << dof));
524 };
