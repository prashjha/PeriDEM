79 static json getExampleJson(const json &given = json::object()) {
80 checkNames(given, {"Force_BC_Sets", "Displacement_BC_Sets", "IC_Sets",
81 "Gravity"});
82
83 const auto nForceSets = given.value("Force_BC_Sets", size_t(0));
84 const auto nDispSets = given.value("Displacement_BC_Sets", size_t(0));
85 const auto nICSets = given.value("IC_Sets", size_t(0));
86 const bool gravityActive = given.find("Gravity") != given.end();
87
88 if (nForceSets + nDispSets + nICSets == 0 and !gravityActive)
89 return json({});

91 auto j = json({});
92
93 if (gravityActive)
94 j["Force_BC"]["Gravity"] = given.at("Gravity");
95
96 if (nForceSets > 0)
97 j["Force_BC"]["Sets"] = nForceSets;
98
99 if (nDispSets > 0)
100 j["Displacement_BC"] = {{"Sets", nDispSets}};
101
102 if (nICSets > 0)
103 j["IC"] = {{"Sets", nICSets}};

105 return j;

111 void readFromJson(const json &j) {
112
113 if (j.empty())
114 return;
115
116 // gravity
117 if (j.find("Force_BC") != j.end()) {
118 if (j.at("Force_BC").find("Gravity") != j.at("Force_BC").end()) {
119 d_gravityActive = true;
120 d_gravity = util::Point(j.at("Force_BC").value("Gravity", std::vector<double>({0., 0., 0.})));
121 }
122 }
123
124 // process Force_BC
125 d_forceDeck.clear();
126 d_dispDeck.clear();
127 d_icDeck.clear();
128 for (auto tag : std::vector<std::string>({"Force_BC", "Displacement_BC", "IC"})) {
129 if (j.find(tag) == j.end())
130 continue;
131
132 auto nSets = j.at(tag).value("Sets", 0);
133
134 for (size_t i = 0; i < nSets; i++) {
135
136 auto set = j.at(tag).at("Set_" + std::to_string(i + 1));
137
138 if (tag == "Force_BC") {
139 inp::BCBaseDeck bcData;
140 bcData.readFromJson(set, tag);
141 d_forceDeck.push_back(bcData);
142 } else if (tag == "Displacement_BC") {
143 inp::BCBaseDeck bcData;
144 bcData.readFromJson(set, tag);
145 d_dispDeck.push_back(bcData);
146 } else if (tag == "IC") {
147 inp::BCBaseDeck icData;
148 icData.readFromJson(set, tag);
149 d_icDeck.push_back(icData);
150 }
151 } // loop over sets
152 } // loop over tag
153 }
