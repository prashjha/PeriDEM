26 * -outputDir is used.
27 * -finalTime <t> integration end time (default 0.002; example_twop_circ_contact uses 0.012).
28 * -numSteps <n> number of steps (default 6000; example uses 36000). dt = finalTime/numSteps.
29 * -requireContact fail if max Damage_Z is 0 (particles never damaged / no contact).
30 * -jha2021Table2 Jha et al. JMPS 2021 Table 2 test 1 (ε_n = 1).
31 * -jha2021Table2Test <n> Table 2 test n = 1..5 (sets time, IC, ε_n, CR, damping).

65// Jha et al. JMPS 2021 Table 2: same R, M1, H0; CR vs ε̄_n (paper Epsilon).
66// The paper's CRs (1, 0.946, 0.893, 0.845, 0.796) came from the earlier 2D
67// state-based model, which was about twice too stiff. The values below are
68// from the corrected model on the same frozen mesh; softer grains stay in
69// contact longer, so damping removes more energy.
70struct Jha2021Table2 {
71 double eps;
72 double cr;
73};
74
75Jha2021Table2 jha2021Table2(int test_id) {
76 static const Jha2021Table2 k[] = {
77 {1.0, 0.996}, {0.95, 0.925}, {0.9, 0.861}, {0.85, 0.801}, {0.8, 0.744}};
78 if (test_id < 1 || test_id > 5)
79 throw std::runtime_error("jha2021Table2Test must be 1..5");
80 return k[test_id - 1];
81}

343class RestitutionProbe : public postprocess::Postprocess {
344public:
345 explicit RestitutionProbe(double H0) : d_H0(H0) {}
346
347 void checkStop(data::ModelData &data) override {
348 postprocess::Postprocess::checkStop(data);
349 if (data.d_particlesListTypeAll.size() < 2)
350 return;
351 const auto *p0 = data.d_particlesListTypeAll[0];
352 const auto *p1 = data.d_particlesListTypeAll[1];
353 const double gap = p0->getXCenter().dist(p1->getXCenter()) -
354 p0->d_geom_p->boundingRadius() -
355 p1->d_geom_p->boundingRadius();
356 if (data.currentStep() % 50 == 0)
357 d_samples.push_back({data.d_time, gap});
358 if (d_done)
359 return;
360 const double leave = 1.0e-5;
361 const double significant = 0.05 * d_H0;
362 if (!d_contacted) {
363 if (gap < d_minGap) {
364 d_minGap = gap;
365 d_tMin = data.d_time;
366 } else if (d_minGap < 0.3 * d_H0 && gap > d_minGap + leave) {
367 d_contacted = true;
368 d_H1 = gap;
369 d_tH1 = data.d_time;
370 }
371 } else if (gap > d_H1) {
372 d_H1 = gap;
373 d_tH1 = data.d_time;
374 } else if (d_H1 > d_minGap + significant && gap < d_H1 - leave) {
375 d_done = true;
376 }
377 }
378
379 bool contacted() const { return d_contacted; }
380 double H1() const { return d_H1; }
381 double minGap() const { return d_minGap; }
382 double tMin() const { return d_tMin; }
383 double tH1() const { return d_tH1; }
384 const std::vector<std::pair<double, double>> &samples() const {
385 return d_samples;
386 }
387 double CR() const {
388 if (d_H0 <= 0. || d_H1 <= 0.)
389 return 0.;
390 return std::sqrt(d_H1 / d_H0);
391 }
392
393private:
394 double d_H0;
395 double d_H1 = 0.;
396 double d_minGap = 1.0e9;
397 double d_tMin = 0.;
398 double d_tH1 = 0.;
399 bool d_contacted = false;
400 bool d_done = false;
401 std::vector<std::pair<double, double>> d_samples;
402};

406class LateralProbe : public postprocess::Postprocess {
407public:
408 explicit LateralProbe(double ic_abs_vx) : d_icAbsVx(std::abs(ic_abs_vx)) {}
409
410 void checkStop(data::ModelData &data) override {
411 postprocess::Postprocess::checkStop(data);
412 if (data.d_particlesListTypeAll.size() < 2)
413 return;
414 const auto *p0 = data.d_particlesListTypeAll[0];
415 const auto *p1 = data.d_particlesListTypeAll[1];
416 const double gap = p0->getXCenter().dist(p1->getXCenter()) -
417 p0->d_geom_p->boundingRadius() -
418 p1->d_geom_p->boundingRadius();
419 if (gap < 0.5 * 0.001)
420 d_contacted = true;
421 const double vx = std::abs(p1->getVCenter().d_x);
422 if (vx > d_maxAbsVx)
423 d_maxAbsVx = vx;
424 if (d_contacted && vx < d_minAbsVxAfterContact)
425 d_minAbsVxAfterContact = vx;
426 }
427 bool contacted() const { return d_contacted; }
428 double maxAbsVx() const { return d_maxAbsVx; }
429 double minAbsVxAfterContact() const {
430 return d_contacted ? d_minAbsVxAfterContact : d_maxAbsVx;
431 }
432 double icAbsVx() const { return d_icAbsVx; }
433
434private:
435 double d_icAbsVx = 0.;
436 double d_maxAbsVx = 0.;
437 double d_minAbsVxAfterContact = 1.0e300;
438 bool d_contacted = false;
439};

442class MpiMetricTsProbe : public postprocess::Postprocess {
443public:
444 MpiMetricTsProbe(std::filesystem::path out_dir, size_t interval)
445 : d_outDir(std::move(out_dir)),
446 d_interval(std::max<size_t>(1, interval)) {
447 if (util::parallel::mpiRank() == 0) {
448 std::filesystem::create_directories(d_outDir / "nodal");
449 d_os.open(d_outDir / "mpi_metric_ts.csv");
450 d_os << "step,t,max_u,com1x,com1y\n";
451 }
452 }
453
454 void checkStop(data::ModelData &data) override {
455 postprocess::Postprocess::checkStop(data);
456 const size_t nstep = data.currentStep();
457 if (nstep % d_interval != 0 && nstep < data.numTimeSteps())
458 return;
459
460 double max_u = 0.;
461 for (const auto &u : data.d_u)
462 max_u = std::max(max_u, u.length());
463 if (util::parallel::mpiSize() > 1)
464 MPI_Allreduce(MPI_IN_PLACE, &max_u, 1, MPI_DOUBLE, MPI_MAX,
465 util::parallel::mpiComm());
466 double com1x = 0., com1y = 0.;
467 if (data.d_particlesListTypeParticle.size() > 1) {
468 const auto c = data.d_particlesListTypeParticle[1]->getXCenter();
469 com1x = c.d_x;
470 com1y = c.d_y;
471 }
472 if (util::parallel::mpiRank() == 0) {
473 d_os << std::format("{},{:.12e},{:.12e},{:.12e},{:.12e}\n", nstep,
474 data.d_time, max_u, com1x, com1y);
475 d_os.flush();
476 }
477 dumpNodalUV(data, nstep);
478 }
479
480private:
481 static bool ownsNode(const data::ModelData &data, size_t i) {
482 const int rank = util::parallel::mpiRank();
483 if (data.d_pdDofMpi) {
484 if (i >= data.d_pdNodePartition.size())
485 return false;
486 return static_cast<int>(data.d_pdNodePartition[i]) == rank;
487 }
488 // Particle-MPI / serial: owned grains; walls only from rank 0 (replicated).
489 for (const auto *p : data.d_particlesListTypeAll) {
490 const size_t i0 = p->d_globStart;
491 const size_t i1 = i0 + p->getNumNodes();
492 if (i < i0 || i >= i1)
493 continue;
494 if (p->isWall())
495 return rank == 0;
496 return particle::isLocallyOwned(*p);
497 }
498 return rank == 0;
499 }
500
501 void dumpNodalUV(data::ModelData &data, size_t nstep) {
502 const size_t n = data.d_u.size();
503 std::vector<double> buf(6 * n, 0.);
504 for (size_t i = 0; i < n; ++i) {
505 if (!ownsNode(data, i))
506 continue;
507 buf[6 * i + 0] = data.d_u[i].d_x;
508 buf[6 * i + 1] = data.d_u[i].d_y;
509 buf[6 * i + 2] = data.d_u[i].d_z;
510 buf[6 * i + 3] = data.d_v[i].d_x;
511 buf[6 * i + 4] = data.d_v[i].d_y;
512 buf[6 * i + 5] = data.d_v[i].d_z;
513 }
514 if (util::parallel::mpiSize() > 1)
515 MPI_Allreduce(MPI_IN_PLACE, buf.data(), static_cast<int>(buf.size()),
516 MPI_DOUBLE, MPI_SUM, util::parallel::mpiComm());
517 if (util::parallel::mpiRank() != 0)
518 return;
519 if (!d_wroteXref) {
520 std::ofstream xr(d_outDir / "nodal" / "x_ref.bin", std::ios::binary);
521 const uint32_t nn = static_cast<uint32_t>(n);
522 xr.write(reinterpret_cast<const char *>(&nn), sizeof(nn));
523 for (size_t i = 0; i < n; ++i) {
524 const double p[3] = {data.d_xRef[i].d_x, data.d_xRef[i].d_y,
525 data.d_xRef[i].d_z};
526 xr.write(reinterpret_cast<const char *>(p), sizeof(p));
527 }
528 d_wroteXref = true;
529 }
530 const auto path =
531 d_outDir / "nodal" / std::format("uv_{:06d}.bin", nstep);
532 std::ofstream os(path, std::ios::binary);
533 const char magic[4] = {'P', 'D', 'U', 'V'};
534 const uint32_t step32 = static_cast<uint32_t>(nstep);
535 const uint32_t nn = static_cast<uint32_t>(n);
536 os.write(magic, 4);
537 os.write(reinterpret_cast<const char *>(&step32), sizeof(step32));
538 os.write(reinterpret_cast<const char *>(&nn), sizeof(nn));
539 os.write(reinterpret_cast<const char *>(buf.data()),
540 static_cast<std::streamsize>(buf.size() * sizeof(double)));
541 }
542
543 std::filesystem::path d_outDir;
544 size_t d_interval;
545 std::ofstream d_os;
546 bool d_wroteXref = false;
547};
