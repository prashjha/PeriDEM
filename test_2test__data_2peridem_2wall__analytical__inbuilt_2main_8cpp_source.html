197struct Probe {
198 size_t n_contact_nodes = 0;
199 double min_gap = 1.e300;
200 double max_force_err = 0.;
201 double max_force_ref = 0.;
202};

204Probe checkWallSpringOnly(PeriDEMModel &dem, geom::GeomObject &wall_geom) {
205 Probe p;
206 if (!dem.d_contact_p || !dem.d_contact_p->d_wallContact ||
207 !dem.d_contact_p->d_pairForce)
208 throw std::runtime_error("analytical wall: contact path not set up");
209
210 const auto &contact = dem.d_particleDeck_p->d_contactDeck.getContact(0, 1);
211 const double Kn = contact.d_Kn;
212 const double Rc = contact.d_contactR;
213
214 for (auto &f : dem.d_f)
215 f = util::Point();
216 dem.d_contact_p->d_wallContact->apply(dem, dem.d_contact_p->d_pairForce.get(),
217 /*use_node_damping*/ false);
218
219 for (size_t i = 0; i < dem.d_x.size(); ++i) {
220 if (dem.d_ptId[i] != 0)
221 continue;
222 geom::WallContactHit hit;
223 if (!wall_geom.wallContactQuery(dem.d_x[i], hit) || !hit.active)
224 continue;
225 if (!(hit.signed_gap < Rc))
226 continue;
227
228 ++p.n_contact_nodes;
229 p.min_gap = std::min(p.min_gap, hit.signed_gap);
230
231 auto scalar = Kn * (hit.signed_gap - Rc) * dem.d_vol[i];
232 if (scalar > 0.)
233 scalar = 0.;
234 const util::Point f_ref = scalar * (-1. * hit.outward_n);
235 const double err = (dem.d_f[i] - f_ref).length();
236 p.max_force_err = std::max(p.max_force_err, err);
237 p.max_force_ref = std::max(p.max_force_ref, f_ref.length());
238 }
239 if (p.n_contact_nodes == 0)
240 p.min_gap = 0.;
241 return p;
242}

244double minGrainWallGap(PeriDEMModel &dem, geom::GeomObject &wall_geom) {
245 double ming = 1.e300;
246 size_t n = 0;
247 for (size_t i = 0; i < dem.d_x.size(); ++i) {
248 if (dem.d_ptId[i] != 0)
249 continue;
250 geom::WallContactHit hit;
251 if (!wall_geom.wallContactQuery(dem.d_x[i], hit) || !hit.active)
252 continue;
253 ming = std::min(ming, hit.signed_gap);
254 ++n;
255 }
256 return n ? ming : 0.;
257}

261int main(int argc, char *argv[]) {
262 util::parallel::initMpi(argc, argv);
263 util::io::InputParser input(argc, argv);
264
265 unsigned n_threads = 2;
266 if (input.cmdOptionExists("-nThreads"))
267 n_threads = static_cast<unsigned>(std::stoi(input.getCmdOption("-nThreads")));
268 util::parallel::initNThreads(n_threads);
269
270 namespace fs = std::filesystem;
271 fs::path base = fs::current_path() / "wall_analytical_run";
272 if (input.cmdOptionExists("-outputDir"))
273 base = input.getCmdOption("-outputDir");
274 const fs::path out_dir = base / "out";
275 const fs::path inp_dir = base / "inp";
276 fs::create_directories(out_dir);
277 fs::create_directories(inp_dir);
278
279 const double R = 0.001;
280 const double mesh_size = R / 5.0;
281 const double horizon = 3.0 * mesh_size;
282 const double Rc_factor = 0.95;
283 const double Kn = 1.0e11;
284 const double final_time = 5.0e-4;
285 const size_t num_steps = 5000;
286
287 const bool policy_combo = input.cmdOptionExists("-policyCombo");
288 auto input_json =
289 buildInputJson(directoryPathWithTrailingSep(out_dir), inp_dir / "mesh_cir.msh",
290 inp_dir / "mesh_wall.msh", R, mesh_size, horizon, Rc_factor, Kn,
291 final_time, num_steps, policy_combo);
292 {
293 std::ofstream os(inp_dir / "input.json");
294 os << input_json.dump(2);
295 }
296
297 // -deckOnly stops after the deck is written. The deck comparison reads the
298 // deck and does not need this driver to run the simulation.
299 if (input.cmdOptionExists("-deckOnly")) {
300 util::io::print("deck written; -deckOnly, not running\n");
301 util::parallel::finalizeMpi();
302 return 0;
303 }
304
305 auto deck = std::make_shared<inp::Input>(input_json);
306 PeriDEMModel dem(deck);
307 dem.init();
308
309 if (dem.d_modelDeck_p->d_wallContact != "analytical_plane")
310 throw std::runtime_error("analytical wall: Wall_Contact not analytical_plane");
311 if (policy_combo) {
312 if (dem.d_modelDeck_p->d_selfContact != "reference_gap")
313 throw std::runtime_error("alternate deck: Self_Contact is not reference_gap");
314 if (dem.d_modelDeck_p->d_bondBreak != "absolute_stretch")
315 throw std::runtime_error("alternate deck: Bond_Break is not absolute_stretch");
316 if (dem.d_particleDeck_p->d_contactDeck.d_frictionLaw != "stick_slip")
317 throw std::runtime_error("alternate deck: Friction_Law is not stick_slip");
318 if (dem.d_particleDeck_p->d_contactDeck.d_dampingLaw != "node")
319 throw std::runtime_error("alternate deck: Damping_Law is not node");
320 if (!dem.d_contact_p || !dem.d_contact_p->d_pairForce ||
321 dem.d_contact_p->d_useNodeDamping != true)
322 throw std::runtime_error("alternate deck: node damping not active on contact");
323 util::io::print("alternate deck fields applied\n");
324 }
325 if (!dem.d_contact_p || !dem.d_contact_p->d_wallContact ||
326 !dem.d_contact_p->d_wallContact->skipsMeshedGrainWall())
327 throw std::runtime_error("analytical wall: must skip meshed grain-wall");
328
329 const double Rc = dem.d_particleDeck_p->d_contactDeck.getContact(0, 1).d_contactR;
330 // Floor top face is at y=0 (matches rectangle placement in the deck).
331 geom::Plane floor_plane(util::Point(0., 1., 0.), util::Point(0., 0., 0.));
332
333 // Wall spring-only force density vs closed form.
334 {
335 const double push = 0.5 * Rc;
336 for (size_t i = 0; i < dem.d_x.size(); ++i) {
337 if (dem.d_ptId[i] != 0)
338 continue;
339 dem.d_u[i].d_y -= push;
340 dem.d_x[i] = dem.d_xRef[i] + dem.d_u[i];
341 dem.d_v[i] = util::Point();
342 }
343 const auto frozen = checkWallSpringOnly(dem, floor_plane);
344 util::io::print(std::format(
345 "analytical wall formula: n_contact={}, min_gap={}, Rc={}, max|f_ref|={}, max|f-f_ref|={}\n",
346 frozen.n_contact_nodes, frozen.min_gap, Rc, frozen.max_force_ref,
347 frozen.max_force_err));
348 if (frozen.n_contact_nodes == 0)
349 throw std::runtime_error("analytical wall: no grain nodes in contact");
350 if (!(frozen.max_force_ref > 0.))
351 throw std::runtime_error("analytical wall: expected nonzero force");
352 if (!(frozen.max_force_err <= 1.e-6 * (1. + frozen.max_force_ref)))
353 throw std::runtime_error(std::format(
354 "analytical wall: force density mismatch vs Kn*(gap-Rc)*voli*(-n): err={}",
355 frozen.max_force_err));
356 if (!(frozen.min_gap > -Rc))
357 throw std::runtime_error(std::format(
358 "analytical wall: deep penetration past Rc in contact config: min_gap={}, Rc={}",
359 frozen.min_gap, Rc));
360 }
361
362 // Full contact assembly (PD + analytical wall) must keep the same gap bound.
363 dem.computeForces();
364 const double ming = minGrainWallGap(dem, floor_plane);
365 util::io::print(std::format("analytical wall after computeForces: min_gap={}, Rc={}\n", ming, Rc));
366 if (!(ming < Rc))
367 throw std::runtime_error(
368 "analytical wall: expected grain-wall contact (min_gap < Rc)");
369 if (!(ming > -Rc))
370 throw std::runtime_error(std::format(
371 "analytical wall: deep penetration past Rc: min_gap={}, Rc={}", ming, Rc));
372
373 dem.close();
374 util::io::print("analytical wall OK\n");
375 return 0;
376}
