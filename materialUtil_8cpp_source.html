23double computeStateMxI(size_t i, const std::vector<util::Point> &nodes,
24 const std::vector<double> &nodal_vol,
25 const std::vector<std::vector<size_t>> &neighbors,
26 const std::vector<std::vector<float>> &neighbors_sq_dist,
27 const double &mesh_size,
28 const material::Material *material) {
29
30 double horizon = material->getHorizon();
31 const auto &xi = nodes[i];
32 double m = 0.;
33
34 // upper and lower bound for volume correction
35 auto check_up = horizon + 0.5 * mesh_size;
36 auto check_low = horizon - 0.5 * mesh_size;
37
38 size_t k = 0;
39 for (size_t j : neighbors[i]) {
40
41 const auto &xj = nodes[j];
42 double rji = (xj - xi).length();
43 //double rji = std::sqrt(neighbors_sq_dist[i][k]);
44
45 if (util::isGreater(rji, horizon)) // or j == i) <-- check! For weighted volume, we don't want to skip the node itself
46 continue;
47
48 // get corrected volume of node j
49 auto volj = nodal_vol[j];
50
51 if (util::isGreater(rji, check_low))
52 volj *= (check_up - rji) / mesh_size;
53
54 m += std::pow(rji, 2) * material->getInfFn(rji) * volj;
55
56 k++;
57 }
58
59 if (util::isLess(m, 1.0E-18)) {
60 std::ostringstream oss;
61 oss << "Error: Weighted nodal volume = " << m
62 << " should not be too close to zero.\n";
63
64 oss << "Mesh size = " << mesh_size << "\n";
65 oss << "J = " << material->getInfFn(0.5 * horizon) << "\n";
66 oss << "weighted volume (times 1.0e10) = " << m*1.e+10 << "\n";
67 oss << "nodal coord = " << xi.printStr() << "\n";
68 oss << material->printStr(0, 0);
69
70 throw std::runtime_error(
71 util::io::Msg()
72 << oss.str());
73 }
74 return m;
75}

77double computeStateThetaxI(size_t i, const std::vector<util::Point> &nodes,
78 const std::vector<util::Point> &nodes_disp,
79 const std::vector<double> &nodal_vol,
80 const std::vector<std::vector<size_t>> &neighbors,
81 const std::vector<std::vector<float>> &neighbors_sq_dist,
82 const double &mesh_size,
83 const material::Material *material,
84 const geometry::Fracture *fracture,
85 const std::vector<double> &mx) {
86
87 double horizon = material->getHorizon();
88 const auto &xi = nodes[i];
89 const auto &ui = nodes_disp[i];
90 double m = mx[i];
91 double theta = 0.;
92
93 // upper and lower bound for volume correction
94 auto check_up = horizon + 0.5 * mesh_size;
95 auto check_low = horizon - 0.5 * mesh_size;
96
97 size_t k = 0;
98 for (size_t j : neighbors[i]) {
99
100 const auto &xj = nodes[j];
101 const auto &uj = nodes_disp[j];
102 double rji = (xj - xi).length();
103 //double rji = std::sqrt(neighbors_sq_dist[i][k]); // distance in
104 // reference configuration
105
106 if (util::isGreater(rji, horizon) or j == i)
107 continue;
108
109 // get corrected volume of node j
110 auto volj = nodal_vol[j];
111
112 if (util::isGreater(rji, check_low))
113 volj *= (check_up - rji) / mesh_size;
114
115 // get bond state
116 double bond_state = fracture->getBondState(i, k) ? 0. : 1.;
117
118 // get change in bond length
119 auto yi = xi + ui;
120 auto yj = xj + uj;
121 double change_length = (yj - yi).length() - rji;
122
123 theta += bond_state * rji * change_length *
124 material->getInfFn(rji) * volj;
125
126 k += 1;
127 }
128
129 return 3. * theta / m;
130}

132double computeHydrostaticStrainI(size_t i, const std::vector<util::Point> &nodes,
133 const std::vector<util::Point> &nodes_disp,
134 const std::vector<double> &nodal_vol,
135 const std::vector<std::vector<size_t>> &neighbors,
136 const std::vector<std::vector<float>> &neighbors_sq_dist,
137 const double &mesh_size,
138 const material::Material *material,
139 const geometry::Fracture *fracture,
140 size_t dim) {
141
142 double horizon = material->getHorizon();
143 const auto &xi = nodes[i];
144 const auto &ui = nodes_disp[i];
145 double theta = 0.;
146
147 // upper and lower bound for volume correction
148 auto check_up = horizon + 0.5 * mesh_size;
149 auto check_low = horizon - 0.5 * mesh_size;
150
151 // get volume of ball
152 double vol_ball = std::pow(horizon, 2) * M_PI;
153 if (dim == 3)
154 vol_ball *= horizon * 4. / 3.;
155
156 size_t k = 0;
157 for (size_t j : neighbors[i]) {
158
159 const auto &xj = nodes[j];
160 const auto &uj = nodes_disp[j];
161 double rji = (xj - xi).length();
162 //double rji = std::sqrt(neighbors_sq_dist[i][k]);
163
164 if (util::isGreater(rji, horizon) or j == i)
165 continue;
166
167 // get corrected volume of node j
168 auto volj = nodal_vol[j];
169
170 if (util::isGreater(rji, check_low))
171 volj *= (check_up - rji) / mesh_size;
172
173 // get bond state
174 double bond_state = fracture->getBondState(i, k) ? 0. : 1.;
175
176 // get bond strain
177 double Sji = material->getS(xj - xi, uj - ui);
178
179 theta += bond_state * rji * Sji *
180 material->getInfFn(rji) * volj / vol_ball;
181
182 k += 1;
183 }
184
185 return theta;
186}

188void updateBondFractureDataI(size_t i, const std::vector<util::Point> &nodes,
189 const std::vector<std::vector<size_t>> &neighbors,
190 const std::vector<util::Point> &nodes_disp,
191 const material::Material *material,
192 geometry::Fracture *fracture) {
193
194 size_t k = 0;
195 for (size_t j : neighbors[i]) {
196
197 double s = material->getS(nodes[j] - nodes[i], nodes_disp[j] -
198 nodes_disp[i]);
199 double sc = material->getSc((nodes[j] - nodes[i]).length());
200
201 // get fracture state, modify, and set
202 auto fs = fracture->getBondState(i, k);
203 if (!fs && util::isGreater(std::abs(s), sc + 1.0e-10))
204 fs = true;
205 fracture->setBondState(i, k, fs);
206
207 k += 1;
208 }
209}

213void material::computeStateMx(data::ModelData *model, bool compute_in_parallel) {
214
215 model->d_mX.resize(model->d_x.size());
216 if (!compute_in_parallel) {
217 for (size_t i = 0; i < model->d_x.size(); i++) {
218 const auto &pti = model->getPtId(i);
219 const auto &particle = model->getParticleFromAllList(pti);
220 auto mx = computeStateMxI(i, model->d_xRef, model->d_vol,
221 model->d_neighPd, model->d_neighPdSqdDist,
222 particle->getMeshSize(),
223 particle->getMaterial());
224
225 model->setMx(i, mx);
226 }
227 } else {
228
229 tf::Executor executor(util::parallel::getNThreads());
230 tf::Taskflow taskflow;
231
232 taskflow.for_each_index(
233 (std::size_t) 0, model->d_x.size(), (std::size_t) 1, [model](std::size_t i) {
234 const auto &pti = model->getPtId(i);
235 const auto &particle = model->getParticleFromAllList(pti);

238 particle->getMeshSize(),
239 particle->getMaterial());
240
241 model->setMx(i, mx);
242 }
243 ); // for_each
244
245 executor.run(taskflow).get();
