23std::shared_ptr<mesh::Mesh>
24mesh_gen::createParticleMesh(const inp::MeshDeck &zmeshDeck,
25 const geom::GeomData &zgeomDeck,
26 const inp::ModelDeck *modelDeck,
27 const std::string &modelName) {
28 if (!zmeshDeck.d_createMesh) {
29 return std::make_shared<mesh::Mesh>(&zmeshDeck, modelDeck);
30 }
31
32 if (zmeshDeck.d_createMeshInfo == "uniform" &&
33 (zgeomDeck.d_geomName == "rectangle" || zgeomDeck.d_geomName == "cuboid" ||
34 zgeomDeck.d_geomName == "square" || zgeomDeck.d_geomName == "cube")) {
35
36 if (!zgeomDeck.d_geom_p)
37 throw std::runtime_error(
38 modelName + ": uniform mesh on a box requires particle geometry object "
39 "(geom::createGeomObject on Particle.Set_i).");
40
41 const size_t dim = modelDeck->d_dim;
42 const auto bb = zgeomDeck.d_geom_p->box();
43 std::pair<std::vector<double>, std::vector<double>> box;
44 box.first.reserve(dim);
45 box.second.reserve(dim);
46 for (size_t i = 0; i < dim; ++i) {
47 const double lo_i =
48 (i == 0) ? bb.first.d_x : (i == 1) ? bb.first.d_y : bb.first.d_z;
49 const double hi_i =
50 (i == 0) ? bb.second.d_x : (i == 1) ? bb.second.d_y : bb.second.d_z;
51 box.first.push_back(lo_i);
52 box.second.push_back(hi_i);
53 }
54
55 std::vector<size_t> nGrid(dim);
56 for (size_t i = 0; i < dim; ++i) {
57 const double span = box.second[i] - box.first[i];
58 if (span <= 0.)
59 throw std::runtime_error(
60 modelName + ": uniform mesh: non-positive axis extent from geom box "
61 "(axis " +
62 std::to_string(i) + ").");
63 nGrid[i] = static_cast<size_t>(span / zmeshDeck.d_hMeshing);
64 }
65
66 mesh::Mesh temp_mesh;
67 mesh::createUniformMesh(&temp_mesh, dim, box, nGrid);
68 if (!zmeshDeck.d_voidRegions.empty())
69 mesh::removeNodesInBoxes(&temp_mesh, zmeshDeck.d_voidRegions);
70 return std::make_shared<mesh::Mesh>(temp_mesh);
71 }
72
73 if (zmeshDeck.d_createMeshInfo == "gmsh_builtin_mesh") {
74 if (!zgeomDeck.d_geom_p)
75 throw std::runtime_error(
76 modelName + ": gmsh_builtin_mesh requires particle geometry object "
77 "(geom::createGeomObject on Particle.Set_i).");
78 mesh::Mesh temp_mesh;
79 const std::string mesh_stem =
80 zmeshDeck.d_filename.empty()
81 ? std::string()
82 : util::io::removeExtensionFromFile(zmeshDeck.d_filename);
83 mesh_gen::generateBuiltinParticleMeshGmsh(
84 zgeomDeck.d_geom_p, zmeshDeck.d_hMeshing, mesh_stem, false,
85 zmeshDeck.d_writeMeshFile, &temp_mesh, &zmeshDeck, modelDeck);
86 return std::make_shared<mesh::Mesh>(temp_mesh);
87 }
88
89 throw std::runtime_error(
90 "Error: Unsupported in-built mesh: CreateMesh.Info = " +
91 zmeshDeck.d_createMeshInfo + " with geometry = " + zgeomDeck.d_geomName);
92}
