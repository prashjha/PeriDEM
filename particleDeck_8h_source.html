95 static json getParticleGeomExampleJson(std::vector<geom::GeomData> pGeomVec = std::vector<geom::GeomData>()) {
96
97 auto nSets = pGeomVec.size();
98
99 if (nSets == 0)
100 return json({});
101
102 auto j = json({{"Sets", nSets}});
103
104 for (size_t i = 0; i < nSets; i++) {
105 auto js = json({});
106 geom::writeGeometry(js, pGeomVec[i]);
107 std::string set_name = "Set_" + std::to_string(i+1);
108 j[set_name] = js;
109 }
110
111 return j;
112 }

117 void readParticleGeomFromJson(const json &j) {
118
119 if (j.empty())
120 return;
121
122 if (d_particleSimType == "Multi_Particle") {
123 auto nSets = j.value("Sets", size_t(0));
124
125 d_pGeomVec.resize(nSets);
126
127 for (size_t i = 0; i < nSets; i++) {
128 auto set_name = "Set_" + std::to_string(i + 1);
129 if (j.find(set_name) == j.end())
130 throw std::runtime_error("Set " + set_name + " not found in particle geometry");
131
132 auto js = j.at(set_name);
133
134 // read
135 geom::readGeometry(js, d_pGeomVec[i]);
136 // create
137 geom::createGeomObject(d_pGeomVec[i]);
138 }
139 } else if (d_particleSimType == "Single_Particle") {
140 d_pGeomVec.resize(1);
141 auto js = j.find("Set_1") == j.end()? j : j.at("Set_1");
142 if (js.find("Type") != js.end()) {
143 geom::readGeometry(js, d_pGeomVec[0]);
144 geom::createGeomObject(d_pGeomVec[0]);
145 }
146 }

153 static json getParticleMeshExampleJson(std::vector<std::string> filenameVec = std::vector<std::string>(),

173 void readParticleMeshFromJson(const json &j) {
174 if (j.empty())
175 return;
176
177 if (d_particleSimType == "Multi_Particle") {
178 auto nSets = j.value("Sets", size_t(0));
179 d_pMeshVec.resize(nSets);
180
181 for (size_t i = 0; i < nSets; i++) {
182 auto set_name = "Set_" + std::to_string(i + 1);
183
184 if (j.find(set_name) == j.end())
185 throw std::runtime_error("Set " + set_name + " not found in particle mesh");
186
187 auto js = j.at(set_name);
188
189 // copy this set from a previous set?
190 int copy_set = js.value("Copy_Data", int(-1));
191 if (copy_set != -1) {
192 std::string copy_set_tag = "Set_" + std::to_string(copy_set);
193 auto js_copy = j.at(copy_set_tag);
194 d_pMeshVec[i].readFromJson(js_copy);
195 } else
196 d_pMeshVec[i].readFromJson(js);
197 }
198 } else if (d_particleSimType == "Single_Particle") {
199 d_pMeshVec.resize(1);
200 auto js = j.find("Set_1") == j.end()? j : j.at("Set_1");
201 d_pMeshVec[0].readFromJson(js);
202 }
203 }
