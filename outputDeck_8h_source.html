168 void readFromJson(const json &j) {
169 if (j.empty())
170 return ;
171
172 d_outFormat = j.value("File_Format", "vtu");
173 d_path = j.value("Path", "./");
174 if (d_path.empty())
175 d_path = "./";
176 d_dtOut = j.value("Output_Interval", size_t(1));
177 if (d_dtOut < 1)
178 throw std::runtime_error(
179 "Output_Interval must be >= 1 (use 1 to write every step).");
180 d_dtOutOld = d_dtOut;
181 d_dtOutCriteria = d_dtOut;
182
183 d_debug = j.value("Debug", size_t(2));
184 d_performFEOut = j.value("Perform_FE_Out", true);
185 d_compressType = j.value("Compress_Type", "zlib");
186 d_performOut = j.value("Perform_Out", true);
187 d_dtTestOut = j.value("Test_Output_Interval", size_t(1));
188 d_tagPPFile = j.value("Tag_PP", "");
189
190 d_pvdCollection = j.value("PVD_Collection", false);
191
192 d_outTags = j.value("Tags", std::vector<std::string>());
193
194 if (j.contains("Output_Criteria")) {
195 d_outCriteria = j.at("Output_Criteria").value("Type", std::string(""));
196 d_dtOutCriteria = j.at("Output_Criteria").value("New_Interval", size_t(1));
197 d_outCriteriaParams = j.at("Output_Criteria").value("Parameters", std::vector<double>());
198 if (d_dtOutCriteria < 1)
199 throw std::runtime_error(
200 "Output_Criteria.New_Interval must be >= 1 (use 1 to write every step).");
201 }
202 }
