208 void readFromJson(const json &j) {
209 if (j.empty())
210 return ;
211
212 readFields(*this, j, fields());
213
214 // An empty path names the working directory.
215 if (d_path.empty())
216 d_path = "./";
217
218 // Both follow the output interval until the criteria block changes one.
219 d_dtOutOld = d_dtOut;
220 d_dtOutCriteria = d_dtOut;
221
222 if (j.contains("Output_Criteria")) {
223 d_outCriteria = j.at("Output_Criteria").value("Type", std::string(""));
224 d_dtOutCriteria = j.at("Output_Criteria").value("New_Interval", size_t(1));
225 d_outCriteriaParams = j.at("Output_Criteria").value("Parameters", std::vector<double>());
226 if (d_dtOutCriteria < 1)
227 throw std::runtime_error(
228 "Output_Criteria.New_Interval must be >= 1 (use 1 to write every step).");
229 }
230 }
