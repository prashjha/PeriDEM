81 static json getExampleJson(const json &given = json::object()) {
82 const std::vector<std::string> names = {
83 "File", "Mesh_Size", "Create_Mesh",
84 "Info", "Write_Mesh_File", "Void_Regions"};
85 checkNames(given, names);
86
87 const auto filename = given.value("File", std::string());
88 const auto h_meshing = given.value("Mesh_Size", -1.);
89 const auto createMeshInfo =
90 given.value("Info", std::string("gmsh_builtin_mesh"));
91 const auto writeMeshFile = given.value("Write_Mesh_File", true);
92 const auto voidRegions = given.value(
93 "Void_Regions", std::vector<std::vector<double>>());
94
95 auto j = json({});
96 if (!filename.empty())
97 j["File"] = filename;
98
99 // A mesh size with no filename requests generation. Create_Mesh lets a
100 // set both generate a mesh and name the file it is written to or read
101 // back from.
102 const bool create = given.value("Create_Mesh", false) ||
103 (filename.empty() && h_meshing > 0.);
104 if (create) {
105 if (!(h_meshing > 0.))
106 throw std::runtime_error(
107 "MeshDeck::getExampleJson: mesh creation needs a positive "
108 "Mesh_Size.");
109 auto cm = applyGiven(json{{"Flag", true},
110 {"Info", createMeshInfo},
111 {"Mesh_Size", h_meshing},
112 {"Write_Mesh_File", writeMeshFile}},
113 createMeshFields(), {"Void_Regions"});
114 for (const auto &r : voidRegions) {
115 if (r.size() != 6)
116 throw std::runtime_error(
117 "MeshDeck::getExampleJson: each Void_Regions entry needs 6 "
118 "values [xlo, ylo, zlo, xhi, yhi, zhi].");
119 }
120 if (!voidRegions.empty())
121 cm["Void_Regions"] = voidRegions;
122 j["CreateMesh"] = cm;
123 } else if (filename.empty()) {
124 throw std::runtime_error(
125 "MeshDeck::getExampleJson: a mesh set needs a File or a Mesh_Size.");
126 }
127 return j;
128 }

178 void readFromJson(const json &j) {
179 if (j.empty())
180 return;
181
182 readFields(*this, j, fields());
183 // Defaults of the sub-block apply whether or not it is present.
184 readFields(*this, json::object(), createMeshFields());
185
186 if (j.find("CreateMesh") != j.end()) {
187 const auto &cm = j.at("CreateMesh");
188 readFields(*this, cm, createMeshFields());
189
190 // A mesh size written beside the block rather than inside it.
191 if (cm.find("Mesh_Size") == cm.end() && d_createMesh &&
192 j.find("Mesh_Size") != j.end())
193 d_hMeshing = j.at("Mesh_Size").get<double>();
194
195 d_voidRegions.clear();
196 if (cm.find("Void_Regions") != cm.end()) {
197 for (const auto &r : cm.at("Void_Regions")) {
198 auto box = r.get<std::vector<double>>();
199 if (box.size() != 6)
200 throw std::runtime_error(
201 "CreateMesh.Void_Regions entries need 6 values [xlo,ylo,zlo,xhi,yhi,zhi].");
202 d_voidRegions.push_back(std::move(box));
203 }
204 }
205 }
206
207 if (d_createMesh && d_hMeshing <= 0.)
208 throw std::runtime_error(
209 "In-built mesh creation requires CreateMesh.Mesh_Size (or legacy top-level Mesh_Size).");
210
211 if (d_filename.empty() && !d_createMesh) {
212 throw std::runtime_error("Mesh filename can not be empty unless CreateMesh is enabled.");
213 }
214 }
