96 void readFromJson(const json &j) {
97 if (j.empty())
98 return;
99
100 d_filename = j.value("File", std::string());
101 d_hMeshing = 0.;
102 d_createMesh = false;
103 d_createMeshInfo = "uniform";
104 d_writeMeshFile = true;
105
106 if (j.find("CreateMesh") != j.end()) {
107 const auto &cm = j.at("CreateMesh");
108 d_createMesh = cm.value("Flag", false);
109 d_createMeshInfo = cm.value("Info", std::string("uniform"));
110 d_writeMeshFile = cm.value("Write_Mesh_File", true);
111 if (cm.find("Mesh_Size") != cm.end())
112 d_hMeshing = cm.at("Mesh_Size").get<double>();
113 else if (d_createMesh && j.find("Mesh_Size") != j.end())
114 d_hMeshing = j.at("Mesh_Size").get<double>();
115
116 d_voidRegions.clear();
117 if (cm.find("Void_Regions") != cm.end()) {
118 for (const auto &r : cm.at("Void_Regions")) {
119 auto box = r.get<std::vector<double>>();
120 if (box.size() != 6)
121 throw std::runtime_error(
122 "CreateMesh.Void_Regions entries need 6 values [xlo,ylo,zlo,xhi,yhi,zhi].");
123 d_voidRegions.push_back(std::move(box));
124 }
125 }
126 }
127
128 if (d_createMesh && d_hMeshing <= 0.)
129 throw std::runtime_error(
130 "In-built mesh creation requires CreateMesh.Mesh_Size (or legacy top-level Mesh_Size).");
131
132 if (d_filename.empty() && !d_createMesh) {
133 throw std::runtime_error("Mesh filename can not be empty unless CreateMesh is enabled.");
134 }
135 }
