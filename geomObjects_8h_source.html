55 inline util::Point mapSimilarity(const util::Point &x, const util::Point &pivotDefault,
56 const util::Point &t, double scale, double angle, const util::Point &axis,
57 const util::Point *rotationPoint) {
58 const util::Point &p = (rotationPoint != nullptr) ? *rotationPoint : pivotDefault;
59 return p + util::rotate(x - p, angle, axis) * scale + t;

65struct BoxPartition {
66
67util::Point d_xc; // centroid of box
68std::pair<util::Point, util::Point> d_box; // two corner points
69double d_r; // radius of circle inscribing box
70std::vector<size_t> d_nodes; // id of nodes in this box
71
72BoxPartition() : d_xc(util::Point()),
73 d_box({util::Point(), util::Point()}),
74 d_r(0.) {}
75
76bool isNear(const BoxPartition &box, const double &tol) const {
77 auto dx = d_xc - box.d_xc;
78 return dx.length() < d_r + box.d_r + tol;
79}
80
81bool isNear(const util::Point &x, const double &tol) const {
82
83 if (util::isLess(x.d_x, d_box.first.d_x - tol) or
84 util::isLess(x.d_y, d_box.first.d_y - tol) or
85 util::isLess(x.d_z, d_box.first.d_z - tol) or
86 util::isGreater(x.d_x, d_box.second.d_x + tol) or
87 util::isGreater(x.d_y, d_box.second.d_y + tol) or
88 util::isGreater(x.d_z, d_box.second.d_z + tol))
89 return false;
90
91 return true;
92}
93
94void addNode(const size_t &i) {
95 for (const auto &j: d_nodes)
96 if (j == i)
97 return;
98
99 d_nodes.push_back(i);
100}
101};
