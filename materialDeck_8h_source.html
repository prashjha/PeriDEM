484 void readFromJson(const json &j) {
485 if (j.empty())
486 return;
487
488 // read mat data
489 d_matData.readFromJson(j);
490
491 readFields(*this, j, fields());
492
493 // Older decks spell this Is_Plain_Strain. The table reads the current
494 // spelling, so the older one is applied when it alone is present.
495 if (j.find("Is_Plane_Strain") == j.end() &&
496 j.find("Is_Plain_Strain") != j.end())
497 d_isPlaneStrain = j.at("Is_Plain_Strain").get<bool>();
498
499 checkGroups(j, groups());
500
501 if (j.find("Influence_Function") != j.end())
502 readFields(*this, j.at("Influence_Function"), influenceFields());
503 if (!d_computeParamsFromElastic && d_bondPotentialParams.empty())
504 throw std::runtime_error(
505 "Compute_From_Classical is false but Bond_Potential_Params is missing.");
506 }
