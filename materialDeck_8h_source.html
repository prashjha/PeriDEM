368 static json getExampleJson(std::string materialType = "PDState", bool isPlainStrain = false,
369 double horizon = -1., double horizonMeshRatio = -1., double density = 1.,
370 double K = 0., double G = 0., double Gc = 0., bool computeParamsFromElastic = true,
371 size_t influenceFnType = 0, double E = -1.) {

381 j["Horizon"] = horizon;
382 if (horizonMeshRatio > 0)
383 j["Horizon_Mesh_Ratio"] = horizonMeshRatio;
384
385 j["Compute_From_Classical"] = computeParamsFromElastic;
386 if (E > 0.)
387 j["E"] = E;
388 j["K"] = K;
389 j["G"] = G;
390 j["Gc"] = Gc;
391
392 j["Influence_Function"] = {{"Type", influenceFnType}};
393
394 return j;

400 void readFromJson(const json &j) {
401 if (j.empty())
402 return;
403
404 // read mat data
405 d_matData.readFromJson(j);
406
407 d_computeParamsFromElastic = j.value("Compute_From_Classical", true);
408 d_isPlaneStrain = j.value("Is_Plain_Strain", false);
409 d_materialType = j.value("Type", "");
410 d_density = j.value("Density", 1.);
411 d_horizon = j.value("Horizon", -1.);
412 d_horizonMeshRatio = j.value("Horizon_Mesh_Ratio", -1.);
413 if (d_horizon < 0. and d_horizonMeshRatio < 0.) {
414 throw std::runtime_error("Horizon and Horizon Mesh Ratio both are invalid.");
415 return;
416 }
417 if (j.find("Influence_Function") != j.end()) {
418 d_influenceFnType = j.at("Influence_Function").value("Type", 0);
419 d_influenceFnParams = j.at("Influence_Function").value("Parameters", std::vector<double>());
420 }
421
422 // Explicit micromodulus / critical stretch, used when Compute_From_Classical
423 // is false (e.g. to follow a specific paper's calibration convention).
424 d_bondPotentialParams =
425 j.value("Bond_Potential_Params", std::vector<double>());
426 d_statePotentialParams =
427 j.value("State_Potential_Params", std::vector<double>());
428 if (!d_computeParamsFromElastic && d_bondPotentialParams.empty())
429 throw std::runtime_error(
430 "Compute_From_Classical is false but Bond_Potential_Params is missing.");
431 }
