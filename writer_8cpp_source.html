30void rw::writer::Writer::open(const std::string &filename,
31 const std::string &format,
32 const std::string &compress_type) {
33 d_format = format;
34 if (d_format == "vtu")
35 d_vtkWriter_p = new rw::writer::VtkWriter(filename, compress_type);
36 else if (d_format == "msh")
37 d_mshWriter_p = new rw::writer::MshWriter(filename, compress_type);
38 else if (d_format == "legacy_vtk")
39 d_legacyVtkWriter_p =
40 new rw::writer::LegacyVtkWriter(filename, compress_type);
41 else {
42 throw std::runtime_error(
43 util::io::Msg()
44 << "File format " << d_format << " is not supported.\n");
45 }
46}
