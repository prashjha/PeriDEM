100 void readFromJson(const json &j) {
101 if (j.empty())
102 return;
103
104 d_dampingLaw = j.value("Damping_Law", "com_and_node");
105 d_frictionLaw = j.value("Friction_Law", "coulomb_simple");
106 d_correctVolume = j.value("Correct_Volume", true);
107
108 if (d_dampingLaw != "com_and_node" && d_dampingLaw != "com" &&
109 d_dampingLaw != "node" && d_dampingLaw != "off") {
110 throw std::runtime_error(
111 "Contact.Damping_Law must be com_and_node|com|node|off.");
112 }
113 if (d_frictionLaw != "coulomb_simple" && d_frictionLaw != "stick_slip") {
114 throw std::runtime_error(
115 "Contact.Friction_Law must be coulomb_simple|stick_slip.");
116 }
117
118 auto nSets = j.value("Sets", size_t(0));
119 d_data.resize(nSets);
120 for (size_t i=0; i<nSets; i++)
121 d_data[i].resize(nSets);
122
123 for (size_t i = 0; i < nSets; i++) {
124 for (size_t k = i; k < nSets; k++) {
125 std::string set_name = "Set_" + std::to_string(i+1) + "_" + std::to_string(k+1);
126 //std::cout << "\n\n\n processing " << set_name << "\n\n\n" << std::flush;
127
128 auto cd = inp::ContactPairDeck();
129
130 if (j.find(set_name) == j.end())
131 throw std::runtime_error("Set " + set_name + " not found in contact block.");
132
133 auto js = j.at(set_name);
134
135 // copy this set from previous set?
136 std::vector<int> copy_set = js.value("Copy_Data", std::vector<int>({-1, -1}));
137 //std::cout << "copy_data = " << copy_set[0] << ", " << copy_set[1] << "\n" << std::flush;
138
139 // check
140 if ((copy_set[0] != -1 and copy_set[1] == -1) or (copy_set[0] == -1 and copy_set[1] != -1) or (copy_set.size() != 2))
141 throw std::runtime_error("The pair of set ids for copying is invalid in contact block.");
142
143 if (copy_set[0] != -1) {
144 std::string copy_set_tag = "Set_" + std::to_string(copy_set[0]) + "_" + std::to_string(copy_set[1]);
145 auto js_copy = j.at(copy_set_tag);
146 cd.readFromJson(js_copy);
147 } else {
148 cd.readFromJson(js);
149 }
150
151 d_data[i][k] = cd;
152 if (i != k)
153 d_data[k][i] = cd;
154 }
155 }
156 }
