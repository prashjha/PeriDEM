137size_t applyPrecrack(PeriDEMModel &dem, double x_crack) {
138 size_t n_broken = 0;
139 for (size_t i = 0; i < dem.d_neighPd.size(); ++i) {
140 const auto &xi = dem.d_xRef[i];
141 for (size_t k = 0; k < dem.d_neighPd[i].size(); ++k) {
142 const size_t j = dem.d_neighPd[i][k];
143 const auto &xj = dem.d_xRef[j];
144 if ((xi.d_x - x_crack) * (xj.d_x - x_crack) < 0.) {
145 dem.d_fracture_p->setBondState(i, k, true);
146 ++n_broken;
147 }
148 }
149 }
150 return n_broken;
151}

153void applyRigidOpenClose(PeriDEMModel &dem, double x_crack, double u_right) {
154 for (size_t i = 0; i < dem.d_xRef.size(); ++i) {
155 const double ux = (dem.d_xRef[i].d_x >= x_crack) ? u_right : 0.;
156 dem.d_u[i] = util::Point(ux, 0., 0.);
157 dem.d_v[i] = util::Point();
158 dem.d_x[i] = dem.d_xRef[i] + dem.d_u[i];
159 }
160}

162struct CrackMetrics {
163 double min_gap = 1.0e300;
164 double max_gap = 0.;
165 size_t n_pairs = 0;
166 double force_x_left = 0.;
167 double force_x_right = 0.;
168};
169
170CrackMetrics measure(PeriDEMModel &dem, double x_crack, double band) {
171 CrackMetrics m;
172 for (size_t i = 0; i < dem.d_neighPd.size(); ++i) {
173 const auto &xi = dem.d_xRef[i];
174 for (size_t k = 0; k < dem.d_neighPd[i].size(); ++k) {
175 if (!dem.d_fracture_p->getBondState(i, k))
176 continue;
177 const size_t j = dem.d_neighPd[i][k];
178 const auto &xj = dem.d_xRef[j];
179 if ((xi.d_x - x_crack) * (xj.d_x - x_crack) >= 0.)
180 continue;
181 const double gap = (dem.d_x[j] - dem.d_x[i]).length();
182 m.min_gap = std::min(m.min_gap, gap);
183 m.max_gap = std::max(m.max_gap, gap);
184 ++m.n_pairs;
185 }
186 }
187 if (m.n_pairs == 0)
188 m.min_gap = 0.;
189
190 for (size_t i = 0; i < dem.d_xRef.size(); ++i) {
191 const double x = dem.d_xRef[i].d_x;
192 const double fx = dem.d_f[i].d_x * dem.d_vol[i];
193 if (x < x_crack && x > x_crack - band)
194 m.force_x_left += fx;
195 if (x >= x_crack && x < x_crack + band)
196 m.force_x_right += fx;
197 }
198 return m;
199}

207LawResult runLaw(const std::string &self_contact, const std::filesystem::path &run_dir, double L,
208 double mesh_size, double horizon, unsigned n_threads,
209 bool deck_only = false) {
210 namespace fs = std::filesystem;
211 const fs::path out_dir = run_dir / "out";
212 const fs::path inp_dir = run_dir / "inp";
213 fs::create_directories(out_dir);
214 fs::create_directories(inp_dir);
215
216 const fs::path mesh = inp_dir / "mesh_rect.msh";
217 // Time fields unused for prescribed kinematics; keep Output_Interval valid.
218 auto input_json =
219 buildInputJson(directoryPathWithTrailingSep(out_dir), mesh, L, mesh_size, horizon, 1.0e-6,
220 2, self_contact);
221 {
222 std::ofstream os(inp_dir / "input.json");
223 os << input_json.dump(2);
224 }
225
226 // -deckOnly stops after the deck is written. The deck comparison reads the
227 // deck and does not need this driver to run the simulation.
228 if (deck_only) {
229 util::io::print("deck written; -deckOnly, not running\n");
230 return LawResult{};
231 }
232
233 auto deck = std::make_shared<inp::Input>(input_json);
234 PeriDEMModel dem(deck);
235 dem.init();
236
237 const double x_crack = 0.5 * L;
238 const size_t n_broken = applyPrecrack(dem, x_crack);
239 if (n_broken == 0)
240 throw std::runtime_error(self_contact + ": precrack broke no bonds");
241
242 const double Rc = dem.d_maxContactR > 0. ? dem.d_maxContactR : 0.95 * mesh_size;
243 const double h = dem.d_hMax > 0. ? dem.d_hMax : mesh_size;
244 util::io::print(std::format("{}: nodes={}, broken_bonds={}, h={}, Rc={}, horizon={}\n",
245 self_contact, dem.d_x.size(), n_broken, h, Rc, horizon));
246
247 // Open: separate faces so current gap >> Rc (no self-contact yet).
248 const double u_open = 2.0 * Rc;
249 applyRigidOpenClose(dem, x_crack, u_open);
250 dem.computeForces();
251 const auto open_m = measure(dem, x_crack, 2.0 * horizon);
252 if (!(open_m.min_gap > 1.5 * Rc))
253 throw std::runtime_error(std::format("{}: open did not separate (min_gap={})", self_contact,
254 open_m.min_gap));
255
256 // Close into the window where reference_gap (natural R = r0) and
257 // broken_bond_kn (natural R = Rc) disagree: Rc < R < typical r0.
258 // Mid-horizon neighbor distances are ~O(horizon) >> Rc = 0.95 h.
259 const double u_close = -0.5 * (Rc + 0.5 * horizon);
260 applyRigidOpenClose(dem, x_crack, u_close);
261 dem.computeForces();
262 const auto close_m = measure(dem, x_crack, 2.0 * horizon);
263 const double force_amp =
264 0.5 * (std::abs(close_m.force_x_left) + std::abs(close_m.force_x_right));
265
266 util::io::print(std::format("{}: open min_gap={}, close min_gap={}, force_amp={}\n", self_contact,
267 open_m.min_gap, close_m.min_gap, force_amp));
268
269 dem.close();
270
271 LawResult r;
272 r.open_min_gap = open_m.min_gap;
273 r.close_min_gap = close_m.min_gap;
274 r.close_force_amp = force_amp;
275 return r;
276}

280int main(int argc, char *argv[]) {
281 util::parallel::initMpi(argc, argv);
282 util::io::InputParser input(argc, argv);
283
284 unsigned int n_threads = 2;
285 if (input.cmdOptionExists("-nThreads"))
286 n_threads = static_cast<unsigned>(std::stoi(input.getCmdOption("-nThreads")));
287 util::parallel::initNThreads(n_threads);
288
289 namespace fs = std::filesystem;
290 fs::path base = fs::current_path() / "self_contact_mode1_run";
291 if (input.cmdOptionExists("-outputDir"))
292 base = input.getCmdOption("-outputDir");
293
294 const double L = 0.002;
295 const double mesh_size = L / 6.0;
296 const double horizon = 3.0 * mesh_size;
297
298 LawResult bb;
299 LawResult rg;
300 try {
301 const bool deck_only = input.cmdOptionExists("-deckOnly");
302 bb = runLaw("broken_bond_kn", base / "broken_bond_kn", L, mesh_size, horizon,
303 n_threads, deck_only);
304 rg = runLaw("reference_gap", base / "reference_gap", L, mesh_size, horizon,
305 n_threads, deck_only);
306 if (deck_only)
307 return EXIT_SUCCESS;
308 } catch (const std::exception &e) {
309 util::io::print(std::format("self-contact Mode-I failed: {}\n", e.what()));
310 return 1;
311 }
312
313 if (!(bb.close_force_amp > 0.) || !(rg.close_force_amp > 0.)) {
314 util::io::print(std::format(
315 "close did not activate self-contact: F_bb={}, F_rg={}\n", bb.close_force_amp,
316 rg.close_force_amp));
317 return 1;
318 }
319
320 const double fmax = std::max(bb.close_force_amp, rg.close_force_amp);
321 const double rel = std::abs(bb.close_force_amp - rg.close_force_amp) / fmax;
322 if (!(rel > 0.05)) {
323 util::io::print(std::format(
324 "self-contact laws did not differ enough under close: F_bb={}, F_rg={}, rel={}\n",
325 bb.close_force_amp, rg.close_force_amp, rel));
326 return 1;
327 }
328
329 util::io::print(std::format(
330 "self-contact Mode-I OK: open gaps bb={}/rg={}, close forces bb={}/rg={}, rel_diff={}\n", bb.open_min_gap,
331 rg.open_min_gap, bb.close_force_amp, rg.close_force_amp, rel));
332 return 0;
333}
