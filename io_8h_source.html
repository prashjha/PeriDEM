366inline std::string checkAndCreateNewFilename(std::string const & filename, std::string filename_ext)
367{
368 auto f_ext = util::io::getExtensionFromFile(filename);
369 std::string f = filename;
370 if (f_ext.empty()) {
371 f = f + "." + filename_ext;
372 }
373 else {
374 if (f_ext != filename_ext) {
375 throw std::runtime_error(
376 util::io::Msg()
377 << "checkAndCreateNewFilename(): Argument filename = "
378 << filename << " has extension = "
379 << f_ext << " which does not match expected extension = "
380 << filename_ext << std::endl);
381 }
382 }
383
384 return f;
385}
