323inline std::string checkAndCreateNewFilename(std::string const & filename, std::string filename_ext)
324{
325 auto f_ext = util::io::getExtensionFromFile(filename);
326 std::string f = filename;
327 if (f_ext.empty()) {
328 f = f + "." + filename_ext;
329 }
330 else {
331 if (f_ext != filename_ext) {
332 std::cerr << "checkAndCreateNewFilename(): Argument filename = "
333 << filename << " has extension = "
334 << f_ext << " which does not match expected extension = "
335 << filename_ext << std::endl;
336 exit(EXIT_FAILURE);
337 }
338 }
339
340 return f;
341}
