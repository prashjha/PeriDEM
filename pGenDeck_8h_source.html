102 void readFromJson(const json &j) {
103 if (j.empty())
104 return;
105 readFields(*this, j, fields());
106
107 // The positions themselves are a block of their own, not a field.
108 if (d_genMethod == "From_File") {
109 if (j.find("Data") == j.end())
110 throw std::runtime_error("Need information inside key Data.");
111
112 d_pGenJson = j.at("Data");
113 }
114 }
115
120 json writeToJson() const {
121 json j = json::object();
122 writeFields(*this, j, fields());
123 if (d_genMethod == "From_File")
124 j["Data"] = d_pGenJson;
125 return j;
126 }

135 std::string printStr(int nt = 0, int lvl = 0) const {
136
137 auto tabS = util::io::getTabS(nt);
138 std::ostringstream oss;
139 oss << tabS << "------- PGenDeck --------" << std::endl << std::endl;
140 printFields(*this, oss, fields(), tabS);
141 size_t nParticles = d_pGenJson.value("N", 0);
142 oss << tabS << "Number of particles in json object = " << nParticles << std::endl;
143 oss << tabS << "Data for first five particles: " << std::endl;
144 if (nParticles > 0) {
145 nParticles = nParticles > 5 ? 5 : nParticles;
146 for (size_t i=0; i< nParticles; i++) {
147 oss << tabS << "Particle number = " << i << std::endl;
148 oss << tabS << d_pGenJson.at(std::to_string(i)) << std::endl;
149 }
150 }
151 oss << tabS << std::endl;
152
153 return oss.str();
