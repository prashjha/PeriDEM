233 static json getExampleJson(const json &given = json::object(),
234 const geom::GeomData *region = nullptr) {
235 const std::vector<std::string> extra = {
236 "Type", "Time_Function", "Spatial_Function", "IC_Type", "IC_Vector"};
237 checkKeys(given, fields(), extra);
238
239 const auto type = given.value("Type", std::string("Force_BC"));
240
241 auto j = json({});
242
243 if (region != nullptr) {
244 auto j_geom = json({});
245 geom::writeGeometry(j_geom, *region);
246 // Must be an object: readFromJson does j.at("Region").at("Geometry").
247 j["Region"] = json{{"Geometry", j_geom}};
248 }
249
250 json fieldsGiven =
251 json{{"Particle_List",
252 given.value("Particle_List", std::vector<size_t>())},
253 {"Particle_Exclude_List",
254 given.value("Particle_Exclude_List", std::vector<size_t>())}};
255 if (type != "IC") {
256 fieldsGiven["Direction"] =
257 given.value("Direction", std::vector<size_t>());
258 fieldsGiven["Zero_Displacement"] =
259 given.value("Zero_Displacement", false);
260 }
261 j.update(applyGiven(fieldsGiven, fields()));
262
263 if (given.find("Time_Function") != given.end())
264 j["Time_Function"] =
265 applyGiven(given.at("Time_Function"), timeFunctionFields());
266
267 if (given.find("Spatial_Function") != given.end())
268 j["Spatial_Function"] =
269 applyGiven(given.at("Spatial_Function"), spatialFunctionFields());
270
271 if (type == "IC") {
272 const auto icType = given.value("IC_Type", std::string());
273 if (icType == "Constant_Velocity")
274 j[icType]["Velocity_Vector"] =
275 given.value("IC_Vector", std::vector<double>());
276 }
277
278 return j;
279 }

294 void readFromJson(const json &j, std::string type) {
295 if (j.empty())
296 return;
297
298 // fix selection type
299 if (j.find("Region") != j.end()) {
300 d_selectionType = "region";
301 d_isRegionActive = true;
302 if (j.find("Particle_List") != j.end())
303 d_selectionType += "_with_include_list";
304
305 if (j.find("Particle_Exclude_List") != j.end())
306 d_selectionType += "_with_exclude_list";
307 } else {
308 d_isRegionActive = false;
309 if (j.find("Particle_List") != j.end())
310 d_selectionType = "particle";
311 }
312
313 if (d_isRegionActive) {
314 geom::readGeometry(j.at("Region").at("Geometry"), d_regionGeomData);
315 // create a geometry object based on the data
316 geom::createGeomObject(d_regionGeomData);
317 }
318
319 readFields(*this, j, fields());
320
321 if (j.find("Time_Function") != j.end())
322 readFields(*this, j.at("Time_Function"), timeFunctionFields());
323
324 if (j.find("Spatial_Function") != j.end())
325 readFields(*this, j.at("Spatial_Function"), spatialFunctionFields());
326
327 if (type != "IC" && j.find("Direction") == j.end())
328 throw std::runtime_error(
329 "Direction must be specified for boundary condition");
330
331 // if it is initial condition
332 if (type == "IC") {
333 if (j.find("Constant_Velocity") != j.end()) {
334 d_icType = "Constant_Velocity";
335 d_icVec = util::Point(j.at("Constant_Velocity").value("Velocity_Vector", std::vector<double>({0., 0., 0.})));
336 }
337 }
338 }
